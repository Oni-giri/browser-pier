Quick start

curl Python MCP 
# 1. start (or attach to) a profile's browser
curl -X POST https://pier.local/api/sessions \
  -H "Authorization: Bearer $PIER_TOKEN" \
  -d '{"profile": "google", "mode": "visual"}'
# → {"id": "s_8f2c", "cdp": "wss://pier.local/cdp/google"}

# 2. stuck? ask a human, send them the link, wait
POST /api/sessions/s_8f2c/handoff {"reason": "sensitive_input", "detail": "Enter the 2FA code"}
# → {"url": "https://pier.local/h/7Kq2…", "expires_in": 900}
GET  /api/handoffs/h_41?wait=900   # → {"status": "done", "note": "used a backup code"}

# 3. done: free the RAM, keep the cookies
curl -X DELETE https://pier.local/api/sessions/s_8f2c -H "Authorization: Bearer $PIER_TOKEN"
